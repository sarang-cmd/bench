# Mechanics
## 2.1 kinematics
v = u + at
s = ut + (1/2)at^2
v^2 = u^2 + 2as
## 2.2 forces
F = ma
W = mg
## 2.3 momentum and impulse
p = mv
F = dp/dt
J = F t = change in p
## 2.4 work, energy, power
W = F s cos t
KE = (1/2)mv^2
PE = mgh
P = W/t = Fv
efficiency = useful/total
## 2.5 circular motion
v = 2 pi r/T = w r
a = v^2/r = w^2 r
F = mv^2/r
## 2.6 Hooke's law
F = kx
E = (1/2)kx^2
# Thermal physics
## 3.1 energy changes
Q = mc dT
Q = mL
## 3.2 ideal gases
pV = nRT
pV = (1/3) N m c^2
E_k = (3/2) k_B T
n = N/N_A
## 3.3 gas laws
p1V1/T1 = p2V2/T2
# Waves
## 4.1 wave basics
v = f L
## 4.2 superposition
two-source interference: path
difference = nL or (n + 1/2)L
## 4.3 standing waves
f_n = n v / (2L) strings
f_n = n v / (4L) pipes closed one end
## 4.4 Doppler
f' = f (v +- u_o)/(v -+ u_s)
## 4.5 diffraction
theta = L/b for the first minimum
## 4.6 refraction
n = c/v
n1 sin t1 = n2 sin t2
# Electricity and magnetism
## 5.1 charge and current
I = q/t
## 5.2 Ohm's law
V = IR
R = rho L / A
## 5.2 series and parallel
R = R1 + R2
1/R = 1/R1 + 1/R2
## 5.3 power
P = VI = I^2 R = V^2/R
## 5.4 magnetic force
F = BIL sin t
F = qvB sin t
## 5.5 electromagnetic induction
flux = BA cos t
emf = -N d(flux)/dt
emf = BLv
## 5.6 transformers
V_s/V_p = N_s/N_p = I_p/I_s
# Circular motion and gravitation
## 6.1 gravitation
F = G m1 m2 / r^2
g = GM/r^2
## 6.2 orbits
v = sqrt(GM/r)
T^2 = (4 pi^2 / GM) r^3
## 6.3 potential
V = -GM/r
escape speed = sqrt(2GM/r)
# Atomic, nuclear and particle physics
## 7.1 photoelectric effect
E = hf
max KE = hf - phi
## 7.2 atomic energy levels
hf = E2 - E1
## 7.3 radioactive decay
dN/dt = -L N
N = N_0 e^(-Lt)
half life T = ln2 / L
## 7.4 mass-energy
E = mc^2
binding energy = mass defect x c^2
## 7.5 standard model
quarks: u, d, s, c, b, t
leptons: e, nu_e, mu, nu_mu, tau
# Energy production
## 8.1 energy sources
power from a flow: P = (1/2) rho A v^3
## 8.2 solar
albedo and the greenhouse effect
intensity = P/(4 pi r^2)
# Relativity and fields (HL)
## 9.1 relativity
t = t_0 / sqrt(1 - v^2/c^2)
L = L_0 sqrt(1 - v^2/c^2)
m = m_0 / sqrt(1 - v^2/c^2)
# Fields (HL)
## 10.1 electric fields
E = F/q
E = kQ/r^2, k = 1/(4 pi e_0)
V = kQ/r
## 10.2 capacitors
C = Q/V
C = e_0 A/d
E = (1/2) QV
discharge: V = V_0 e^(-t/RC)
## 10.3 magnetic fields
B = mu_0 I/(2 pi r)
# Electromagnetic induction (HL)
## 11.1 inductance
emf = -L dI/dt
E = (1/2) L I^2
# Quantum and nuclear physics (HL)
## 12.1 quantum
de Broglie: L = h/p
uncertainty: dx dp >= h/(4 pi)
## 12.2 nuclear
activity A = L N
mass defect dm = Z m_p + (A-Z)m_n - m_nucleus
